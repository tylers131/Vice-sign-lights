# Dependencies. All pure-Python except dbus-fast, which bleak pulls in.
#
# scripts/install.sh picks the right build automatically:
#
#   armv6  (Pi Zero / Zero W)  -> SKIP_CYTHON=1, pure Python. No compiled wheel
#                                 exists for armv6 anywhere, and building it
#                                 takes the better part of an hour on that core.
#   armv7  (Pi Zero 2 W, Pi 3) -> piwheels serves a compiled wheel.
#   arm64  (Pi 4/5, 64-bit OS) -> PyPI manylinux serves a compiled wheel.
#
# Installing by hand on a Zero W? Prefix it: SKIP_CYTHON=1 pip install -r ...
# Anywhere else, plain pip is correct and faster.

bleak>=0.22,<4           # BLE client
Flask>=2.2,<4            # pure Python
waitress>=2.1,<4         # pure-Python WSGI server, steadier than the dev server

# OPTIONAL, deliberately not listed above: the DHT temperature sensor.
#
#   pip install adafruit-circuitpython-dht
#

# perfectly well without -- vicelights/thermometer.py imports the driver inside
# the read, so everything installs and tests without it. If reads then fail
